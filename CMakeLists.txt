cmake_minimum_required(VERSION 3.20)
project(MedHIL VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
include(CTest)
find_package(Eigen3 3.3 QUIET NO_MODULE)
if(NOT Eigen3_FOUND)
  find_package(Eigen3 5 QUIET NO_MODULE)
endif()
if(NOT Eigen3_FOUND)
  # Eigen is header-only. Avoid configuring its benchmarks and optional backends.
  FetchContent_Declare(eigen GIT_REPOSITORY https://gitlab.com/libeigen/eigen.git
    GIT_TAG 3.4.0 GIT_SHALLOW TRUE SOURCE_SUBDIR medhil_headers_only)
  FetchContent_MakeAvailable(eigen)
  add_library(medhil_eigen INTERFACE)
  target_include_directories(medhil_eigen SYSTEM INTERFACE ${eigen_SOURCE_DIR})
  add_library(Eigen3::Eigen ALIAS medhil_eigen)
endif()
option(MEDHIL_ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_library(medhil_options INTERFACE)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(medhil_options INTERFACE -Wall -Wextra -Wpedantic -Werror)
  if(MEDHIL_ENABLE_SANITIZERS)
    target_compile_options(medhil_options INTERFACE -fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer)
    target_link_options(medhil_options INTERFACE -fsanitize=address,undefined)
  endif()
endif()
add_library(medhil_core STATIC src/registration.cpp src/arm.cpp src/devices.cpp src/safety.cpp src/controller.cpp src/world.cpp)
target_include_directories(medhil_core PUBLIC include)
target_link_libraries(medhil_core PUBLIC Eigen3::Eigen medhil_options)
add_executable(medhil_cli apps/medhil_cli.cpp)
target_link_libraries(medhil_cli PRIVATE medhil_core)
if(BUILD_TESTING)
  FetchContent_Declare(googletest GIT_REPOSITORY https://github.com/google/googletest.git GIT_TAG v1.14.0 GIT_SHALLOW TRUE)
  set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
  FetchContent_MakeAvailable(googletest)
  add_executable(medhil_tests tests/math.cpp tests/safety.cpp tests/devices.cpp tests/scenarios.cpp)
  target_link_libraries(medhil_tests PRIVATE medhil_core GTest::gtest_main)
  include(GoogleTest)
  gtest_discover_tests(medhil_tests DISCOVERY_MODE PRE_TEST)
endif()
